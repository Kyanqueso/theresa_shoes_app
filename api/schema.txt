// Theresa Shoes — database schema
// Paste into https://dbdiagram.io to visualize

Table devices {
  id uuid [pk, default: `gen_random_uuid()`]
  device_token varchar [not null, unique]
  label varchar
  is_active boolean [not null, default: true]
  created_at timestamp [not null, default: `now()`]
  last_seen_at timestamp
  failed_pin_attempts integer [not null, default: 0]
  pin_locked_until timestamp [note: 'set after 10 wrong PINs, locks this device out for 30 minutes']
}

Table admin_pin {
  id integer [pk]
  pin_hash varchar [not null]
  updated_at timestamp [not null, default: `now()`]
}

Table companies {
  id uuid [pk, default: `gen_random_uuid()`]
  name varchar [not null, note: 'unique case-insensitively — see ix_companies_name_lower']
  status company_status [not null, default: 'active']
  created_at timestamp [not null, default: `now()`]
  archived_at timestamp [note: 'set when status transitions to archive']
}

Table shoes {
  id uuid [pk, default: `gen_random_uuid()`]
  name varchar [not null]
  price numeric(10,2) [not null, note: 'catalog list price']
  description text
  is_hidden boolean [not null, default: false]
  created_at timestamp [not null, default: `now()`]
}

Table shoe_images {
  id uuid [pk, default: `gen_random_uuid()`]
  shoe_id uuid [not null]
  image_url varchar [not null]
  sort_order integer [not null, default: 0]
}

Table shoe_attribute_options {
  id uuid [pk, default: `gen_random_uuid()`]
  category attribute_category [not null]
  name varchar [not null]
  swatch_color varchar [note: 'materials only (swatch rows)']
  image_url varchar [note: 'mold_type / heel_type / buckle / slingback / flatform']
  is_active boolean [not null, default: true]
  parent_id uuid [note: 'materials only: null = group (e.g. "Helga"), set = swatch belonging to that group']
  created_at timestamp [not null, default: `now()`]
}

Table orders {
  id uuid [pk, default: `gen_random_uuid()`]
  company_id uuid [note: 'nullable — order may not be tied to a tracked company']
  client_name varchar [not null]
  contact_number varchar
  shoe_id uuid [note: 'nullable — manually-entered orders may not reference a catalog shoe']
  custom_model_name varchar [note: 'used instead of shoe_id for manually-entered orders']
  material_id uuid
  mold_type_id uuid
  heel_type_id uuid
  color_code varchar
  size integer
  heel_size integer
  quantity integer [not null, default: 1]
  with_buckle boolean [not null, default: false]
  buckle_id uuid
  with_flatform boolean [not null, default: false]
  flatform_id uuid
  with_slingback boolean [not null, default: false]
  slingback_id uuid
  unit_price numeric(10,2) [not null, note: 'custom per order, defaults from shoes.price']
  status order_status [not null, default: 'current']
  notes_blocks jsonb [not null, default: '[]', note: 'ordered text/photo/drawing/selection blocks, e.g. [{type, value}]']
  created_at timestamp [not null, default: `now()`]
  completed_at timestamp [note: 'set when status transitions to completed, preserved through archiving']
  archived_at timestamp [note: 'set when status transitions to archived']
  archived_with_company boolean [not null, default: false, note: 'true only if archived via the parent company being archived']
}

Table payments {
  id uuid [pk, default: `gen_random_uuid()`]
  order_id uuid [not null, unique]
  client_name varchar [note: 'denormalized from orders for display']
  total_amount numeric(10,2) [not null]
  first_payment numeric(10,2) [not null, default: 0]
  second_payment numeric(10,2) [not null, default: 0]
  third_payment numeric(10,2) [not null, default: 0]
  balance numeric(10,2) [not null]
  balance_cleared_date date
  date_delivered date
}

Enum company_status {
  active
  archive
}

Enum order_status {
  current
  completed
  archived
}

Enum attribute_category {
  material
  mold_type
  heel_type
  buckle
  slingback
  flatform
}
